function drawScene() { actx.clearRect(0, 0, animCanvas.width, animCanvas.height); animTime += 0.04; let w = animCanvas.width; let h = animCanvas.height; if (w === 0 || h === 0) { requestAnimationFrame(drawScene); return; } // Tanah & Latar Belakang Hijau Hutan actx.fillStyle = '#4e9350'; actx.fillRect(0, h - 22, w, 22); // Pokok Melaka di sebelah kiri let treeX = 75; let treeY = h - 22; actx.fillStyle = '#6d4c41'; actx.fillRect(treeX - 12, treeY - 75, 24, 78); actx.fillStyle = '#2e7d32'; actx.beginPath(); actx.arc(treeX, treeY - 80, 42, 0, Math.PI * 2); actx.fill(); // Animasi Anjing Pemburu let dogX = w * 0.52 + Math.sin(animTime * 1.5) * 6; let dogY = h - 38; actx.fillStyle = '#8d6e63'; actx.fillRect(dogX, dogY, 34, 18); actx.fillStyle = '#6d4c41'; actx.fillRect(dogX - 8, dogY - 6, 14, 14); // Kesan Tenangan if (Math.sin(animTime * 3) > 0) { actx.fillStyle = '#ffeb3b'; actx.beginPath(); actx.arc(dogX - 2, dogY + 6, 12 + Math.sin(animTime*5)*4, 0, Math.PI * 2); actx.fill(); } // Animasi Pelanduk Putih let jumpCycle = Math.sin(animTime * 3); let deerX = w * 0.42; let deerY = h - 45 - Math.abs(jumpCycle) * 18; actx.fillStyle = '#ffffff'; actx.strokeStyle = '#90a4ae'; actx.lineWidth = 2; actx.beginPath(); actx.arc(deerX, deerY, 14, 0, Math.PI * 2); actx.fill(); actx.stroke(); actx.beginPath(); actx.arc(deerX - 12, deerY - 8, 9, 0, Math.PI * 2); actx.fill(); actx.stroke(); actx.strokeStyle = '#ffffff'; actx.lineWidth = 4; actx.beginPath(); actx.moveTo(deerX + 4, deerY + 6); actx.lineTo(dogX - 4, dogY + 2); actx.stroke(); // Teks Tajuk Animasi actx.fillStyle = '#1b5e20'; actx.font = 'bold 13px Georgia'; actx.fillText('⚡ Aksi Dramatik: Pelanduk Putih Menendang Anjing Pemburu!', 15, 22); requestAnimationFrame(drawScene); }
